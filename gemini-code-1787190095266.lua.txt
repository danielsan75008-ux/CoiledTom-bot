local function safeParent(inst, parent)
	local ok = pcall(function() inst.Parent = parent end)
	return ok
end

local function markOurs(inst)
	pcall(function() inst:SetAttribute("VC_Added", true) end)
	return inst
end

local function isOurs(inst)
	local ok, v = pcall(function() return inst:GetAttribute("VC_Added") end)
	return ok and v == true
end

local function isOurDummy(inst)
	local m = inst
	while m do
		if m:IsA("Model") and m:GetAttribute("VC_Dummy") then return true end
		m = m.Parent
	end
	return false
end

local function rootHeightAboveGround(model)
	local root = model:FindFirstChild("HumanoidRootPart")
	if not root then return 0 end
	local hum = model:FindFirstChildOfClass("Humanoid")

	if hum and hum.RigType == Enum.HumanoidRigType.R6 then
		return (root.Size.Y / 2) + 2
	end

	if hum then
		return hum.HipHeight + (root.Size.Y / 2)
	end

	local lowest
	for _, p in ipairs(model:GetDescendants()) do
		if p:IsA("BasePart") and BODY_PARTS[p.Name] then
			local bottom = p.Position.Y - (p.Size.Y / 2)
			if not lowest or bottom < lowest then lowest = bottom end
		end
	end
	if lowest then return root.Position.Y - lowest end
	return root.Size.Y / 2
end

local function attachAccessory(char, hum, acc)
	local handle = acc:FindFirstChild("Handle") or acc:FindFirstChildWhichIsA("BasePart")
	if not handle then return false end

	for _, d in ipairs(handle:GetChildren()) do
		if d:IsA("Weld") or d:IsA("Motor6D") or d:IsA("WeldConstraint") then
			pcall(function() d:Destroy() end)
		end
	end

	handle.CanCollide = false
	handle.CanTouch   = false
	handle.CanQuery   = false
	handle.Massless   = true
	pcall(function() handle.Anchored = false end)

	acc.Archivable = true
	if not safeParent(acc, char) then return false end

	pcall(function() hum:AddAccessory(acc) end)

	for _, d in ipairs(handle:GetChildren()) do
		if d:IsA("Weld") and d.Part1 and d.Part1.Parent == char then return true end
	end

	local accAtt
	for _, d in ipairs(handle:GetChildren()) do
		if d:IsA("Attachment") then accAtt = d break end
	end

	if accAtt then
		local bodyAtt
		for _, part in ipairs(char:GetChildren()) do
			if part:IsA("BasePart") then
				local found = part:FindFirstChild(accAtt.Name)
				if found and found:IsA("Attachment") then bodyAtt = found break end
			end
		end

		if bodyAtt then
			local weld = Instance.new("Weld")
			weld.Name   = "VC_AccessoryWeld"
			weld.Part0  = bodyAtt.Parent
			weld.Part1  = handle
			weld.C0     = bodyAtt.CFrame
			weld.C1     = accAtt.CFrame
			weld.Parent = handle
			handle.CFrame = bodyAtt.WorldCFrame * accAtt.CFrame:Inverse()
			return true
		end
	end

	local head = char:FindFirstChild("Head")
	if head then
		local weld = Instance.new("Weld")
		weld.Name   = "VC_AccessoryWeld"
		weld.Part0  = head
		weld.Part1  = handle
		weld.C0     = CFrame.new(0, 0.5, 0)
		local okPos, pos = pcall(function() return acc.AttachmentPos end)
		weld.C1     = CFrame.new(okPos and pos or Vector3.new(0, 0, 0))
		weld.Parent = handle
		return true
	end

	return false
end

local function hasLayeredClothing(accessory)
	for _, d in ipairs(accessory:GetDescendants()) do
		if d:IsA("WrapLayer") then return true end
	end
	return false
end

local function readAnimationPack(appearance, rigName)
	local folder = appearance:FindFirstChild(rigName == "R6" and "R6Anim" or "R15Anim")
	if not folder then return nil end
	local out = {}
	for _, sub in ipairs(folder:GetChildren()) do
		local first = sub:FindFirstChildOfClass("Animation")
		if first and first.AnimationId ~= "" then
			out[string.lower(sub.Name)] = first.AnimationId
		end
	end
	return out
end

	return {

		connections  = {},
		renderBinds  = {},
		stashed      = {},
		added        = {},

		decalSnapshot= {},

		nameTag      = nil,
	}
end

local function snapshotAppearanceValues(state, char)
	for _, obj in ipairs(char:GetDescendants()) do
		if isOurDummy(obj) then
		elseif obj:IsA("BasePart") then
			local entry = {
				part         = obj,
				color        = obj.Color,
				transparency = obj.Transparency,
				ltm          = obj.LocalTransparencyModifier,
			}
			table.insert(state.partSnapshot, entry)
			state.origValues[obj] = entry
		elseif obj:IsA("Decal") or obj:IsA("Texture") then
			local entry = {
				decal        = obj,
				transparency = obj.Transparency,
			}
			table.insert(state.decalSnapshot, entry)
			state.origValues[obj] = entry
		end
	end
end

local function stripAppearance(state, char)
	for _, obj in ipairs(char:GetDescendants()) do
		if not isOurDummy(obj) and not isOurs(obj)
			and (APPEARANCE_CLASSES[obj.ClassName] or obj:IsA("Accoutrement")) then
			table.insert(state.stashed, { inst = obj, parent = obj.Parent })
			safeParent(obj, STASH)
		end
	end

	local head = char:FindFirstChild("Head")
	if head then
		for _, child in ipairs(head:GetChildren()) do
			if not isOurs(child)
				and (child:IsA("Decal") or child:IsA("Texture") or child:IsA("DataModelMesh")) then
				table.insert(state.stashed, { inst = child, parent = head })
				safeParent(child, STASH)
			end
		end
	end
end

local function setCharacterHidden(state, char, hidden)
	for _, obj in ipairs(char:GetDescendants()) do
		if isOurDummy(obj) then
		elseif obj:IsA("BasePart") then
			if hidden then
				if state.origValues[obj] == nil then
					local entry = {
						part = obj, color = obj.Color,
						transparency = obj.Transparency,
						ltm = obj.LocalTransparencyModifier,
					}
					table.insert(state.partSnapshot, entry)
					state.origValues[obj] = entry
				end
				obj.LocalTransparencyModifier = 1
				obj.Transparency = 1
			else
				local entry = state.origValues[obj]
				obj.LocalTransparencyModifier = entry and entry.ltm or 0
				obj.Transparency              = entry and entry.transparency or 0
			end
		elseif obj:IsA("Decal") or obj:IsA("Texture") then
			if hidden then
				if state.origValues[obj] == nil then
					local entry = { decal = obj, transparency = obj.Transparency }
					table.insert(state.decalSnapshot, entry)
					state.origValues[obj] = entry
				end
				obj.Transparency = 1
			else
				local entry = state.origValues[obj]
				obj.Transparency = entry and entry.transparency or 0
			end
		end
	end

	local head = char:FindFirstChild("Head")
	if head then
		local bb = head:FindFirstChildOfClass("BillboardGui")
		if bb then
			if hidden and not state.nameTag then
				state.nameTag = { gui = bb, enabled = bb.Enabled }
			end
			bb.Enabled = not hidden
		end
	end
end

local function restorePlayer(player)
	local state = states[player]
	if not state then return end
	states[player] = nil

	for _, conn in ipairs(state.connections) do
		pcall(function() conn:Disconnect() end)
	end
	for _, name in ipairs(state.renderBinds) do
		pcall(function() RunService:UnbindFromRenderStep(name) end)
	end

	if state.dummy then
		pcall(function() state.dummy:Destroy() end)
	end

	for _, inst in ipairs(state.added) do
		if inst and inst.Parent then
			pcall(function() inst:Destroy() end)
		end
	end

	local returnedAccessories = {}
	for _, entry in ipairs(state.stashed) do
		if entry.inst and entry.inst.Parent == STASH and entry.parent and entry.parent.Parent then
			if safeParent(entry.inst, entry.parent) and entry.inst:IsA("Accoutrement") then
				table.insert(returnedAccessories, { acc = entry.inst, parent = entry.parent })
			end
		end
	end

	for _, item in ipairs(returnedAccessories) do
		local acc    = item.acc
		local handle = acc:FindFirstChild("Handle") or acc:FindFirstChildWhichIsA("BasePart")
		if handle then
			local welded = false
			for _, d in ipairs(handle:GetChildren()) do
				if (d:IsA("Weld") or d:IsA("WeldConstraint")) and d.Part0 and d.Part1 then
					welded = true break
				end
			end
			if not welded then
				local owner   = item.parent
				local ownHum  = owner and owner:FindFirstChildOfClass("Humanoid")
				if ownHum then
					pcall(function() attachAccessory(owner, ownHum, acc) end)
				end
			end
		end
	end

	for _, snap in ipairs(state.partSnapshot) do
		local p = snap.part
		if p and p.Parent then
			pcall(function()
				p.Color                     = snap.color
				p.Transparency              = snap.transparency
				p.LocalTransparencyModifier = snap.ltm
			end)
		end
	end
	for _, snap in ipairs(state.decalSnapshot) do
		local d = snap.decal
		if d and d.Parent then
			pcall(function() d.Transparency = snap.transparency end)
		end
	end

	for _, snap in ipairs(state.animSnapshot) do
		if snap.anim and snap.anim.Parent then
			pcall(function() snap.anim.AnimationId = snap.id end)
		end
	end

	if state.nameTag and state.nameTag.gui and state.nameTag.gui.Parent then
		pcall(function() state.nameTag.gui.Enabled = state.nameTag.enabled end)
	end
end

local function restoreAll()
	for player in pairs(states) do
		restorePlayer(player)
	end
	for _, child in ipairs(STASH:GetChildren()) do
		pcall(function() child:Destroy() end)
	end
end

local function buildR6Rig()
	local model = Instance.new("Model")
	model.Name = "R6Rig"

	local function newPart(name, size)
		local p = Instance.new("Part")
		p.Name         = name
		p.Size         = size
		p.Color        = Color3.fromRGB(163, 162, 165)
		p.Material     = Enum.Material.Plastic
		p.TopSurface   = Enum.SurfaceType.Smooth
		p.BottomSurface= Enum.SurfaceType.Smooth
		p.CanCollide   = false
		p.CanQuery     = false
		p.CanTouch     = false
		p.Massless     = true
		p.Anchored     = false
		p.Parent       = model
		return p
	end

	local function newAttachment(parent, name, pos)
		local a = Instance.new("Attachment")
		a.Name     = name
		a.Position = pos
		a.Parent   = parent
		return a
	end

	local hrp   = newPart("HumanoidRootPart", Vector3.new(2, 2, 1))
	hrp.Transparency = 1
	local torso = newPart("Torso",     Vector3.new(2, 2, 1))
	local head  = newPart("Head",      Vector3.new(2, 1, 1))
	local la    = newPart("Left Arm",  Vector3.new(1, 2, 1))
	local ra    = newPart("Right Arm", Vector3.new(1, 2, 1))
	local ll    = newPart("Left Leg",  Vector3.new(1, 2, 1))
	local rl    = newPart("Right Leg", Vector3.new(1, 2, 1))

	local headMesh = Instance.new("SpecialMesh")
	headMesh.MeshType = Enum.MeshType.Head
	headMesh.Scale    = Vector3.new(1.25, 1.25, 1.25)
	headMesh.Parent   = head

	model.PrimaryPart = hrp

	hrp.CFrame   = CFrame.new(0, 3, 0)
	torso.CFrame = CFrame.new(0, 3, 0)
	head.CFrame  = CFrame.new(0, 4.5, 0)
	la.CFrame    = CFrame.new(-1.5, 3, 0)
	ra.CFrame    = CFrame.new( 1.5, 3, 0)
	ll.CFrame    = CFrame.new(-0.5, 1, 0)
	rl.CFrame    = CFrame.new( 0.5, 1, 0)

	newAttachment(hrp,   "RootAttachment",       Vector3.new(0, 0, 0))
	newAttachment(head,  "HatAttachment",        Vector3.new(0, 0.6, 0))
	newAttachment(head,  "HairAttachment",       Vector3.new(0, 0.6, 0))
	newAttachment(head,  "FaceFrontAttachment",  Vector3.new(0, 0, -0.6))
	newAttachment(head,  "FaceCenterAttachment", Vector3.new(0, 0, 0))
	newAttachment(head,  "NeckAttachment",       Vector3.new(0, -0.5, 0))
	newAttachment(torso, "NeckAttachment",       Vector3.new(0, 1, 0))
	newAttachment(torso, "BodyFrontAttachment",  Vector3.new(0, 0, -0.5))
	newAttachment(torso, "BodyBackAttachment",   Vector3.new(0, 0, 0.5))
	newAttachment(torso, "LeftCollarAttachment", Vector3.new(-1, 1, 0))
	newAttachment(torso, "RightCollarAttachment",Vector3.new(1, 1, 0))
	newAttachment(torso, "WaistCenterAttachment",Vector3.new(0, -1, 0))
	newAttachment(torso, "WaistFrontAttachment", Vector3.new(0, -1, -0.5))
	newAttachment(torso, "WaistBackAttachment",  Vector3.new(0, -1, 0.5))
	newAttachment(la,    "LeftShoulderAttachment",  Vector3.new(0, 1, 0))
	newAttachment(la,    "LeftGripAttachment",      Vector3.new(0, -1, 0))
	newAttachment(ra,    "RightShoulderAttachment", Vector3.new(0, 1, 0))
	newAttachment(ra,    "RightGripAttachment",     Vector3.new(0, -1, 0))
	newAttachment(ll,    "LeftFootAttachment",      Vector3.new(0, -1, 0))
	newAttachment(rl,    "RightFootAttachment",     Vector3.new(0, -1, 0))

	local HALF_PI = math.pi / 2
	local function newMotor(name, parent, p0, p1, c0, c1)
		local m = Instance.new("Motor6D")
		m.Name   = name
		m.Part0  = p0
		m.Part1  = p1
		m.C0     = c0
		m.C1     = c1
		m.Parent = parent
		return m
	end

	newMotor("RootJoint", hrp, hrp, torso,
		CFrame.new(0, 0, 0) * CFrame.Angles(-HALF_PI, 0, math.pi),
		CFrame.new(0, 0, 0) * CFrame.Angles(-HALF_PI, 0, math.pi))

	newMotor("Neck", torso, torso, head,
		CFrame.new(0, 1, 0)    * CFrame.Angles(-HALF_PI, 0, math.pi),
		CFrame.new(0, -0.5, 0) * CFrame.Angles(-HALF_PI, 0, math.pi))

	newMotor("Right Shoulder", torso, torso, ra,
		CFrame.new(1, 0.5, 0)    * CFrame.Angles(0, HALF_PI, 0),
		CFrame.new(-0.5, 0.5, 0) * CFrame.Angles(0, HALF_PI, 0))

	newMotor("Left Shoulder", torso, torso, la,
		CFrame.new(-1, 0.5, 0)  * CFrame.Angles(0, -HALF_PI, 0),
		CFrame.new(0.5, 0.5, 0) * CFrame.Angles(0, -HALF_PI, 0))

	newMotor("Right Hip", torso, torso, rl,
		CFrame.new(1, -1, 0)   * CFrame.Angles(0, HALF_PI, 0),
		CFrame.new(0.5, 1, 0)  * CFrame.Angles(0, HALF_PI, 0))

	newMotor("Left Hip", torso, torso, ll,
		CFrame.new(-1, -1, 0)  * CFrame.Angles(0, -HALF_PI, 0),
		CFrame.new(-0.5, 1, 0) * CFrame.Angles(0, -HALF_PI, 0))

	hum.HealthDisplayType   = Enum.HumanoidHealthDisplayType.AlwaysOff
	hum.NameDisplayDistance = 0
	hum.BreakJointsOnDeath  = false
	hum.RequiresNeck        = false
	hum.Parent              = model

	local animator = Instance.new("Animator")
	animator.Parent = hum

	return model
end

local KEEP_LEG_MESHES = true

local function isLegSlot(bodyPart)
	return bodyPart == Enum.BodyPart.LeftLeg or bodyPart == Enum.BodyPart.RightLeg
end

local function r6MeshAllowed(bodyPart)
	if bodyPart == Enum.BodyPart.Head then return true end
	return KEEP_LEG_MESHES and isLegSlot(bodyPart)
end

local function applyAppearance(target, appearance, track, isR6, complement)
	local head    = target:FindFirstChild("Head")
	local destHum = target:FindFirstChildOfClass("Humanoid")

	local have = { acc = {}, Shirt = false, Pants = false, ShirtGraphic = false,
	               BodyColors = false, face = false }
	if complement then
		for _, obj in ipairs(target:GetChildren()) do
			if obj:IsA("Accoutrement") then
				have.acc[string.lower(obj.Name)] = true
			elseif obj:IsA("Shirt") then        have.Shirt        = true
			elseif obj:IsA("Pants") then        have.Pants        = true
			elseif obj:IsA("ShirtGraphic") then have.ShirtGraphic = true
			elseif obj:IsA("BodyColors") then   have.BodyColors   = true
			end
		end
		if head then
			for _, obj in ipairs(head:GetChildren()) do
				if obj:IsA("Decal") or obj:IsA("Texture") then have.face = true end
			end
		end
	end

	local incomingMeshSlots = {}
	if isR6 then
		for _, obj in ipairs(appearance:GetChildren()) do
			if obj:IsA("CharacterMesh") and r6MeshAllowed(obj.BodyPart) then
				incomingMeshSlots[obj.BodyPart] = true
			end
		end
		for _, obj in ipairs(target:GetChildren()) do
			if obj:IsA("CharacterMesh") and incomingMeshSlots[obj.BodyPart] then
				pcall(function() obj:Destroy() end)
			end
		end
	end

	for _, obj in ipairs(appearance:GetChildren()) do
		local class = obj.ClassName

		if obj:IsA("Accoutrement") then
			if isR6 and hasLayeredClothing(obj) then
			elseif complement and have.acc[string.lower(obj.Name)] then
			else
				markOurs(obj)
				if destHum and attachAccessory(target, destHum, obj) then
					table.insert(track, obj)
				else
					obj.Archivable = true
					if safeParent(obj, target) then table.insert(track, obj) end
				end
			end

		elseif class == "CharacterMesh" then
			if isR6 and r6MeshAllowed(obj.BodyPart) then
				obj.Archivable = true
				markOurs(obj)
				if safeParent(obj, target) then table.insert(track, obj) end
			end

		elseif APPEARANCE_CLASSES[class] then
			if complement and have[class] then
			else
				obj.Archivable = true
				markOurs(obj)
				if safeParent(obj, target) then table.insert(track, obj) end
			end

		elseif head and (obj:IsA("Decal") or obj:IsA("Texture")) then
			if complement and have.face then
			else
				obj.Archivable = true
				if safeParent(obj, head) then table.insert(track, obj) end
			end

		elseif head and obj:IsA("DataModelMesh") and not complement then
			for _, existing in ipairs(head:GetChildren()) do
				if existing:IsA("DataModelMesh") then
					pcall(function() existing:Destroy() end)
				end
			end
			obj.Archivable = true
			if safeParent(obj, head) then table.insert(track, obj) end
		end
	end
end

local function driveDummyAnimations(state, realChar, realHum, realRoot, dummy, dummyRigName, animPack)
	local hum = dummy:FindFirstChildOfClass("Humanoid")
	if not hum then return end
	local animator = hum:FindFirstChildOfClass("Animator") or Instance.new("Animator", hum)

	local set = {}
	for k, v in pairs(ANIM_SETS[dummyRigName]) do set[k] = v end
	if animPack then
		for k, v in pairs(animPack) do
			if set[k] then set[k] = v end
		end
	end

	local tracks = {}
	for key, id in pairs(set) do
		local anim = Instance.new("Animation")
		anim.AnimationId = id
		local ok, track = pcall(function() return animator:LoadAnimation(anim) end)
		if ok and track then
			track.Looped   = (key ~= "jump")
			track.Priority = (key == "idle") and Enum.AnimationPriority.Core or Enum.AnimationPriority.Movement
			tracks[key] = track
		end
	end

	local current = nil
	local function switchTo(key)
		if current == key then return end
		if current and tracks[current] then tracks[current]:Stop(0.15) end
		if tracks[key] then tracks[key]:Play(0.15) end
		current = key
	end
	switchTo("idle")

	local bindName = "VC_Anim_" .. tostring(dummy:GetAttribute("VC_Id"))
	table.insert(state.renderBinds, bindName)

	RunService:BindToRenderStep(bindName, Enum.RenderPriority.Character.Value + 2, function()
		if not (realChar.Parent and realHum.Parent and realRoot.Parent and dummy.Parent) then return end

		local st = realHum:GetState()
		local vel = realRoot.AssemblyLinearVelocity or realRoot.Velocity
		local speed = (vel * Vector3.new(1, 0, 1)).Magnitude

		local desired = "idle"
		if st == Enum.HumanoidStateType.Seated or realHum.Sit then
			desired = "sit"
		elseif st == Enum.HumanoidStateType.Climbing then
			desired = "climb"
		elseif st == Enum.HumanoidStateType.Swimming then
			desired = "swim"
		elseif st == Enum.HumanoidStateType.Jumping then
			desired = "jump"
		elseif st == Enum.HumanoidStateType.Freefall then
			desired = "fall"
		elseif speed > 0.75 then
			desired = (speed > 12 and tracks.run) and "run" or "walk"
		end

		switchTo(desired)

		if (current == "walk" or current == "run" or current == "climb") and tracks[current] then
			tracks[current]:AdjustSpeed(math.clamp(speed / 16, 0.35, 3))
		end
	end)
end

local function followRoot(state, realRoot, fakeRoot, deltaY, dummy)
	local offset   = Vector3.new(0, deltaY, 0)
	local bindName = "VC_Follow_" .. tostring(dummy:GetAttribute("VC_Id"))
	table.insert(state.renderBinds, bindName)

	RunService:BindToRenderStep(bindName, Enum.RenderPriority.Camera.Value - 1, function()
		if not (dummy.Parent and realRoot.Parent and fakeRoot.Parent) then return end
		fakeRoot.CFrame = realRoot.CFrame + offset
	end)
end

local function syncMotors(state, realChar, dummy)
	local pairsList = {}
	local fakeMotors = {}
	for _, d in ipairs(dummy:GetDescendants()) do
		if d:IsA("Motor6D") then fakeMotors[d.Name] = d end
	end
	for _, d in ipairs(realChar:GetDescendants()) do
		if d:IsA("Motor6D") and not isOurDummy(d) then
			local fake = fakeMotors[d.Name]
			if fake then table.insert(pairsList, { real = d, fake = fake }) end
		end
	end

	local bindName = "VC_Sync_" .. tostring(dummy:GetAttribute("VC_Id"))
	table.insert(state.renderBinds, bindName)

	RunService:BindToRenderStep(bindName, Enum.RenderPriority.Character.Value + 1, function()
		if not dummy.Parent then return end
		for _, pairEntry in ipairs(pairsList) do
			local r, f = pairEntry.real, pairEntry.fake
			if r.Parent and f.Parent then
				f.Transform = r.Transform
			end
		end
	end)
end

local function httpGetJson(url)
	local HttpService = game:GetService("HttpService")
	local ok, body = pcall(function()
		return game:HttpGet(url, true)
	end)
	if not ok or type(body) ~= "string" then return nil end

	local ok2, decoded = pcall(function() return HttpService:JSONDecode(body) end)
	if ok2 then return decoded end
	return nil
end

local function descriptionFromWeb(userId)
	local data = httpGetJson("https://avatar.roblox.com/v1/users/" .. tostring(userId) .. "/avatar")
	if type(data) ~= "table" then return nil end

	local desc = Instance.new("HumanoidDescription")

	local acc = {}
	for _, asset in ipairs(data.assets or {}) do
		local tid  = asset.assetType and asset.assetType.id
		local prop = tid and WEB_ASSET_MAP[tid]
		if prop and asset.id then
			acc[prop] = acc[prop] and (acc[prop] .. "," .. tostring(asset.id)) or tostring(asset.id)
		end
	end
	for prop, value in pairs(acc) do
		pcall(function() desc[prop] = value end)
	end

	local s = data.scales
	if type(s) == "table" then
		pcall(function()
			desc.HeightScale     = s.height     or 1
			desc.WidthScale      = s.width      or 1
			desc.DepthScale      = s.depth      or 1
			desc.HeadScale       = s.head       or 1
			desc.BodyTypeScale   = s.bodyType   or 0
			desc.ProportionScale = s.proportion or 0
		end)
	end

	local bc = data.bodyColors
	if type(bc) == "table" then
		local function toColor(id)
			local okc, col = pcall(function() return BrickColor.new(id).Color end)
			return okc and col or nil
		end
		pcall(function()
			desc.HeadColor     = toColor(bc.headColorId)     or desc.HeadColor
			desc.TorsoColor    = toColor(bc.torsoColorId)    or desc.TorsoColor
			desc.LeftArmColor  = toColor(bc.leftArmColorId)  or desc.LeftArmColor
			desc.RightArmColor = toColor(bc.rightArmColorId) or desc.RightArmColor
			desc.LeftLegColor  = toColor(bc.leftLegColorId)  or desc.LeftLegColor
			desc.RightLegColor = toColor(bc.rightLegColorId) or desc.RightLegColor
		end)
	end

	return desc
end

local function loadAssetContainer(assetId)
	assetId = tonumber(assetId)
	if not assetId or assetId <= 0 then return nil end

	local InsertService = game:GetService("InsertService")
	local ok, container = pcall(function() return InsertService:LoadAsset(assetId) end)
	if ok and container then return container end

	local ok2, container2 = pcall(function()
		return game:GetService("AssetService"):LoadAsset(assetId)
	end)
	if ok2 and container2 then return container2 end

	return nil
end

local function buildSourceFromDescription(desc)
	if not desc then return nil end

	local src = Instance.new("Model")
	src.Name = "VC_WebSource"
	local got = false

	local bodySlots = { "Torso", "RightArm", "LeftArm", "RightLeg", "LeftLeg", "Head" }
	for _, prop in ipairs(bodySlots) do
		local id = 0
		pcall(function() id = tonumber(desc[prop]) or 0 end)
		local container = loadAssetContainer(id)
		if container then
			for _, obj in ipairs(container:GetDescendants()) do
				if obj:IsA("CharacterMesh") then
					obj.Archivable = true
					obj.Parent = src
					got = true
				end
			end
			pcall(function() container:Destroy() end)
		end
	end

	local accProps = {
		"HatAccessory", "HairAccessory", "FaceAccessory", "NeckAccessory",
		"ShoulderAccessory", "FrontAccessory", "BackAccessory", "WaistAccessory",
	}
	for _, prop in ipairs(accProps) do
		local raw = ""
		pcall(function() raw = tostring(desc[prop] or "") end)
		for idStr in string.gmatch(raw, "%d+") do
			local container = loadAssetContainer(idStr)
			if container then
				for _, obj in ipairs(container:GetDescendants()) do
					if obj:IsA("Accoutrement") then
						obj.Archivable = true
						obj.Parent = src
						got = true
					end
				end
				pcall(function() container:Destroy() end)
			end
		end
	end

	for _, prop in ipairs({ "Shirt", "Pants", "GraphicTShirt" }) do
		local id = 0
		pcall(function() id = tonumber(desc[prop]) or 0 end)
		local container = loadAssetContainer(id)
		if container then
			for _, obj in ipairs(container:GetDescendants()) do
				if obj:IsA("Shirt") or obj:IsA("Pants") or obj:IsA("ShirtGraphic") then
					obj.Archivable = true
					obj.Parent = src
					got = true
				end
			end
			pcall(function() container:Destroy() end)
		end
	end

	local faceId = 0
	pcall(function() faceId = tonumber(desc.Face) or 0 end)
	if faceId > 0 then
		local container = loadAssetContainer(faceId)
		if container then
			local head = Instance.new("Part")
			head.Name = "Head"
			for _, obj in ipairs(container:GetDescendants()) do
				if obj:IsA("Decal") or obj:IsA("Texture") or obj:IsA("DataModelMesh") then
					obj.Archivable = true
					obj.Parent = head
					got = true
				end
			end
			if #head:GetChildren() > 0 then head.Parent = src else head:Destroy() end
			pcall(function() container:Destroy() end)
		end
	end

	pcall(function()
		local bc = Instance.new("BodyColors")
		bc.HeadColor3     = desc.HeadColor
		bc.TorsoColor3    = desc.TorsoColor
		bc.LeftArmColor3  = desc.LeftArmColor
		bc.RightArmColor3 = desc.RightArmColor
		bc.LeftLegColor3  = desc.LeftLegColor
		bc.RightLegColor3 = desc.RightLegColor
		bc.Parent = src
		got = true
	end)

	if not got then src:Destroy() return nil end
	return src
end

local function isRealRig(model)
	if not model then return false end
	return model:FindFirstChildOfClass("Humanoid") ~= nil
	   and model:FindFirstChild("HumanoidRootPart") ~= nil
end

local function rigNameOf(model)
	local h = model and model:FindFirstChildOfClass("Humanoid")
	if not h then return nil end
	return (h.RigType == Enum.HumanoidRigType.R6) and "R6" or "R15"
end

local function fetchAvatarModel(userId, rigName)
	local rigEnum = (rigName == "R6") and Enum.HumanoidRigType.R6 or Enum.HumanoidRigType.R15

	local desc = fetchDescription(userId)
	if desc then
		local ok, model = pcall(function()
			return Players:CreateHumanoidModelFromDescriptionAsync(desc, rigEnum)
		end)
		if ok and isRealRig(model) and rigNameOf(model) == rigName then
			return model, desc, false
		end
		if model then pcall(function() model:Destroy() end) end

		ok, model = pcall(function()
			return Players:CreateHumanoidModelFromDescription(desc, rigEnum)
		end)
		if ok and isRealRig(model) and rigNameOf(model) == rigName then
			return model, desc, false
		end
		if model then pcall(function() model:Destroy() end) end
	end

	if rigName == "R15" then
		local ok, model = pcall(function()
			return Players:CreateHumanoidModelFromUserIdAsync(userId)
		end)
		if ok and isRealRig(model) and rigNameOf(model) == "R15" then
			return model, desc, false
		end
		if model then pcall(function() model:Destroy() end) end

		ok, model = pcall(function()
			return Players:CreateHumanoidModelFromUserId(userId)
		end)
		if ok and isRealRig(model) and rigNameOf(model) == "R15" then
			return model, desc, false
		end
		if model then pcall(function() model:Destroy() end) end
	end

	if desc then
		local built = buildSourceFromDescription(desc)
		if built then return built, desc, true end
	end

	return nil, desc, false
end

local function buildR6FromDescription(desc)
	local rig = buildR6Rig()
	rig.Parent = workspace
	local hum = rig:FindFirstChildOfClass("Humanoid")
	if not hum then rig:Destroy() return nil end

	local ok = pcall(function() hum:ApplyDescriptionAsync(desc) end)
	if not ok then
		ok = pcall(function() hum:ApplyDescription(desc) end)
	end
	if not ok then
		rig:Destroy()
		return nil
	end
	return rig
end

local R6_DESC_SLOTS = {
	{ prop = "Head",     part = Enum.BodyPart.Head     },
	{ prop = "RightLeg", part = Enum.BodyPart.RightLeg },
	{ prop = "LeftLeg",  part = Enum.BodyPart.LeftLeg  },
}

local function stripBodyMeshes(state, target)
	for _, obj in ipairs(target:GetChildren()) do
		if obj:IsA("CharacterMesh") then
			if not r6MeshAllowed(obj.BodyPart) then
				table.insert(state.stashed, { inst = obj, parent = target })
				safeParent(obj, STASH)
			end
		end
	end
end

local function fillMissingR6Meshes(state, target, desc, usedSlots)
	if not desc then return end
	local InsertService = game:GetService("InsertService")

	for _, slot in ipairs(R6_DESC_SLOTS) do
		if not usedSlots[slot.part] then
			local assetId = 0
			pcall(function() assetId = tonumber(desc[slot.prop]) or 0 end)

			if assetId > 0 then
				local ok, container = pcall(function()
					return InsertService:LoadAsset(assetId)
				end)

				if ok and container then
					for _, obj in ipairs(container:GetDescendants()) do
						if obj:IsA("CharacterMesh") and obj.BodyPart == slot.part then
							for _, old in ipairs(target:GetChildren()) do
								if old:IsA("CharacterMesh") and old.BodyPart == slot.part then
									table.insert(state.stashed, { inst = old, parent = target })
									safeParent(old, STASH)
								end
							end
							obj.Archivable = true
							if safeParent(obj, target) then
								table.insert(state.added, obj)
								usedSlots[slot.part] = true
							end
							break
						end
					end
					pcall(function() container:Destroy() end)
				end
			end
		end
	end
end

local function applyNative(player, char, hum, userId, setStatus)
	local state = makeState(player)
	state.mode  = "Native"
	states[player] = state

	snapshotAppearanceValues(state, char)

	stripAppearance(state, char)
	setCharacterHidden(state, char, true)
	setStatus("Baixando aparência...")

	local r6Source, r6Desc = fetchAvatarModel(userId, "R6")
	local usedSlots = {}
	if r6Source then
		r6Source.Parent = STASH

		for _, obj in ipairs(r6Source:GetChildren()) do
			if obj:IsA("CharacterMesh") then
				for _, old in ipairs(char:GetChildren()) do
					if old:IsA("CharacterMesh") and old.BodyPart == obj.BodyPart then
						table.insert(state.stashed, { inst = old, parent = char })
						safeParent(old, STASH)
					end
				end
				obj.Archivable = true
				if safeParent(obj, char) then table.insert(state.added, obj) end
				usedSlots[obj.BodyPart] = true

			elseif obj:IsA("Accoutrement") then
				if not hasLayeredClothing(obj) then
					markOurs(obj)
					if attachAccessory(char, hum, obj) then table.insert(state.added, obj) end
				end

			elseif obj:IsA("Shirt") or obj:IsA("Pants") or obj:IsA("ShirtGraphic")
				or obj:IsA("BodyColors") then
				obj.Archivable = true
				markOurs(obj)
				if safeParent(obj, char) then table.insert(state.added, obj) end
			end
		end

		local srcHead = r6Source:FindFirstChild("Head")
		local dstHead = char:FindFirstChild("Head")
		if srcHead and dstHead then
			for _, existing in ipairs(dstHead:GetChildren()) do
				if existing:IsA("DataModelMesh") or existing:IsA("Decal") or existing:IsA("Texture") then
					table.insert(state.stashed, { inst = existing, parent = dstHead })
					safeParent(existing, STASH)
				end
			end
			for _, obj in ipairs(srcHead:GetChildren()) do
				if obj:IsA("DataModelMesh") or obj:IsA("Decal") or obj:IsA("Texture") then
					obj.Archivable = true
					if safeParent(obj, dstHead) then table.insert(state.added, obj) end
				end
			end
			if srcHead:IsA("BasePart") and dstHead:IsA("BasePart") then
				dstHead.Transparency = srcHead.Transparency
			end
		end
	end

	fillMissingR6Meshes(state, char, r6Desc or fetchDescription(userId), usedSlots)

	stripBodyMeshes(state, char)

	local appearance = fetchAppearance(userId)
	if not appearance and not r6Source then
		setCharacterHidden(state, char, false)
		restorePlayer(player)
		return false, "Falhou: executor sem HttpGet/InsertService. Nenhuma fonte de aparência respondeu."
	end
	if appearance then
		appearance.Parent = STASH
		applyAppearance(char, appearance, state.added, true, r6Source ~= nil)
	end

	local animate = char:FindFirstChild("Animate")
	local pack = appearance and readAnimationPack(appearance, "R6") or nil
	if animate and pack then
		for _, folder in ipairs(animate:GetChildren()) do
			local id = pack[string.lower(folder.Name)]
			if id then
				for _, a in ipairs(folder:GetChildren()) do
					if a:IsA("Animation") then
						table.insert(state.animSnapshot, { anim = a, id = a.AnimationId })
						a.AnimationId = id
					end
				end
			end
		end
	end

	task.wait(0.15)
	setCharacterHidden(state, char, false)

	local dstHead = char:FindFirstChild("Head")
	if r6Source and dstHead and dstHead:IsA("BasePart") then
		local srcHead = r6Source:FindFirstChild("Head")
		if srcHead and srcHead:IsA("BasePart") and srcHead.Transparency >= 1 then
			dstHead.Transparency              = 1
			dstHead.LocalTransparencyModifier = 0
		end
	end

	if appearance then pcall(function() appearance:Destroy() end) end
	if r6Source then pcall(function() r6Source:Destroy() end) end

	return true
end

local function applyDummy(player, char, hum, userId, wantRig, setStatus)
	local state = makeState(player)
	state.mode  = "Dummy"
	states[player] = state

	snapshotAppearanceValues(state, char)

	setStatus("Montando rig " .. wantRig .. "...")

	local dummy, appearance

	local desc, isBag
	dummy, desc, isBag = fetchAvatarModel(userId, wantRig)

	if dummy and isBag then
		local bag = dummy
		dummy = nil

		if wantRig == "R6" then
			setStatus("Usando fallback de rig R6...")
			local rig = (desc and buildR6FromDescription(desc)) or buildR6Rig()
			if rig then
				rig.Parent = STASH
				applyAppearance(rig, bag, state.added, true, false)
				dummy = rig
			end
		end
		pcall(function() bag:Destroy() end)
	end

	if not dummy and wantRig == "R6" and desc then
		setStatus("Usando fallback de rig R6...")
		dummy = buildR6FromDescription(desc)
	end

	if not dummy then
		restorePlayer(player)
		if wantRig == "R15" then
			return false, "Não consegui um rig R15 real (executor bloqueou a API). Tente o botão R6."
		end
		return false, "Falha ao baixar o avatar (API de aparência bloqueada?)"
	end

	local gotRig = rigNameOf(dummy) or "R15"
	if gotRig ~= wantRig and wantRig == "R6" and desc then
		local rebuilt = buildR6FromDescription(desc)
		if rebuilt then
			pcall(function() dummy:Destroy() end)
			dummy = rebuilt
			gotRig = "R6"
		end
	end

	if wantRig == "R15" and gotRig ~= "R15" then
		pcall(function() dummy:Destroy() end)
		restorePlayer(player)
		return false, "O avatar do alvo voltou como R6; não vou entregar um R15 quadrado. Use o botão R6."
	end

	if wantRig == "R6" then
		local slots = {}
		for _, obj in ipairs(dummy:GetChildren()) do
			if obj:IsA("CharacterMesh") then slots[obj.BodyPart] = true end
		end
		fillMissingR6Meshes(state, dummy, desc or fetchDescription(userId), slots)
		stripBodyMeshes(state, dummy)
	end

	appearance = fetchAppearance(userId)
	if appearance then
		appearance.Parent = STASH
		applyAppearance(dummy, appearance, state.added, wantRig == "R6", true)
	end

	if desc and not dummy:FindFirstChildOfClass("BodyColors") then
		pcall(function()
			local bc = Instance.new("BodyColors")
			bc.HeadColor3     = desc.HeadColor
			bc.TorsoColor3    = desc.TorsoColor
			bc.LeftArmColor3  = desc.LeftArmColor
			bc.RightArmColor3 = desc.RightArmColor
			bc.LeftLegColor3  = desc.LeftLegColor
			bc.RightLegColor3 = desc.RightLegColor
			markOurs(bc)
			bc.Parent = dummy
		end)
	end

	if dummy.Parent == nil then
		dummy.Parent = workspace
	end

	dummyCounter = dummyCounter + 1
	dummy:SetAttribute("VC_Dummy", true)
	dummy:SetAttribute("VC_Id", dummyCounter)
	dummy.Name = "VC_Rig_" .. tostring(dummyCounter)
	state.dummy = dummy

	local fakeHum = dummy:FindFirstChildOfClass("Humanoid")
	if fakeHum then
		fakeHum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
		fakeHum.HealthDisplayType   = Enum.HumanoidHealthDisplayType.AlwaysOff
		fakeHum.NameDisplayDistance = 0
		fakeHum.BreakJointsOnDeath  = false
		fakeHum.RequiresNeck        = false
		fakeHum.PlatformStand       = true
		fakeHum.WalkSpeed           = 0
		fakeHum.JumpPower           = 0
		fakeHum.AutoRotate          = false

		pcall(function() fakeHum.EvaluateStateMachine = false end)

		pcall(function()
			for _, st in ipairs(Enum.HumanoidStateType:GetEnumItems()) do
				if st ~= Enum.HumanoidStateType.None then
					fakeHum:SetStateEnabled(st, false)
				end
			end
			fakeHum:ChangeState(Enum.HumanoidStateType.Physics)
		end)
	end

	for _, d in ipairs(dummy:GetDescendants()) do
		if d:IsA("BasePart") then
			d.CanCollide = false
			d.CanTouch   = false
			d.CanQuery   = false
			d.Massless   = true
			d.Anchored   = false
		elseif d:IsA("BaseScript") then
			pcall(function() d:Destroy() end)
		end
	end

	setCharacterHidden(state, char, true)

	local realRoot = char:FindFirstChild("HumanoidRootPart")
	local fakeRoot = dummy:FindFirstChild("HumanoidRootPart")
	if not (realRoot and fakeRoot) then
		restorePlayer(player)
		return false, "HumanoidRootPart não encontrado"
	end

	local realHeight = rootHeightAboveGround(char)
	local fakeHeight = rootHeightAboveGround(dummy)
	local deltaY     = fakeHeight - realHeight

	dummy.Parent = char
	task.wait()

	fakeRoot.CFrame = realRoot.CFrame + Vector3.new(0, deltaY, 0)

	fakeRoot.Anchored = true
	followRoot(state, realRoot, fakeRoot, deltaY, dummy)

	local realRig = (hum.RigType == Enum.HumanoidRigType.R6) and "R6" or "R15"
	if realRig == wantRig then
		syncMotors(state, char, dummy)
	else
		local pack = appearance and readAnimationPack(appearance, wantRig) or nil
		driveDummyAnimations(state, char, hum, realRoot, dummy, wantRig, pack)
	end

	if appearance then pcall(function() appearance:Destroy() end) end
	return true
end

local function processCloning(mode, setStatus, setButtonText)
	if busy then
		setStatus("Aguarde, já tem uma troca rodando...")
		return
	end
	if not selectedPlayer or not selectedPlayer.Parent then
		setStatus("Selecione um alvo primeiro.")
		return
	end

	local char = selectedPlayer.Character
	local hum  = char and char:FindFirstChildOfClass("Humanoid")
	if not (char and hum) then
		setStatus("O alvo não tem personagem carregado.")
		return
	end

	busy = true
	setButtonText("Carregando...")

	task.spawn(function()
		local finalOk, finalMsg = false, nil

		local runOk, runErr = pcall(function()
			restorePlayer(selectedPlayer)
			task.wait(0.05)

			local username = UsernameInputText and UsernameInputText() or ""
			if username == "" then username = selectedPlayer.Name end

			local idOk, userId = pcall(function()
				return Players:GetUserIdFromNameAsync(username)
			end)
			if not idOk or not userId then
				finalMsg = "Usuário inválido: " .. tostring(username)
				return
			end

			local isR6Game = (hum.RigType == Enum.HumanoidRigType.R6)

			if mode == "R6" and isR6Game then
				finalOk, finalMsg = applyNative(selectedPlayer, char, hum, userId, setStatus)
			else
				finalOk, finalMsg = applyDummy(selectedPlayer, char, hum, userId, mode, setStatus)
			end
		end)

		if not runOk then
			finalOk  = false
			finalMsg = "Erro: " .. tostring(runErr)
			pcall(function() restorePlayer(selectedPlayer) end)
		end

		if finalOk then
			setButtonText("Sucesso!")
			setStatus(("Aplicado %s em %s."):format(mode, selectedPlayer.Name))
		else
			setButtonText("Falhou!")
			setStatus(finalMsg or "Erro desconhecido")
		end

		busy = false
		task.delay(2.5, function() setButtonText(nil) end)
	end)
end

Players.PlayerRemoving:Connect(function(player)
	restorePlayer(player)
end)

local function hookCharacter(player)
	player.CharacterAdded:Connect(function()
		local state = states[player]
		if state then
			for _, conn in ipairs(state.connections) do pcall(function() conn:Disconnect() end) end
			for _, name in ipairs(state.renderBinds) do pcall(function() RunService:UnbindFromRenderStep(name) end) end
			if state.dummy then pcall(function() state.dummy:Destroy() end) end
			states[player] = nil
		end
	end)
end

for _, p in ipairs(Players:GetPlayers()) do hookCharacter(p) end
Players.PlayerAdded:Connect(hookCharacter)

local function updateDropdown()
	for _, child in ipairs(DropdownScroll:GetChildren()) do
		if child:IsA("TextButton") then child:Destroy() end
	end

	local y = 0
	for _, player in ipairs(Players:GetPlayers()) do
		local btn = Instance.new("TextButton")
		btn.Size                   = UDim2.new(1, -8, 0, 28)
		btn.BackgroundColor3       = Color3.fromRGB(24, 24, 24)
		btn.BackgroundTransparency = 0.1
		btn.TextColor3             = Color3.new(1, 1, 1)
		btn.Text                   = player.Name .. (player == LocalPlayer and "  (você)" or "")
		btn.Font                   = Enum.Font.Gotham
		btn.TextSize               = 12
		btn.ZIndex                 = 22
		btn.Parent                 = DropdownScroll
		Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 4)

		btn.Activated:Connect(function()
			selectedPlayer          = player
			DropdownBtn.Text        = "Alvo: " .. player.Name
			DropdownScroll.Visible  = false
			DropdownBackdrop.Visible= false

			local ch  = player.Character
			local hm  = ch and ch:FindFirstChildOfClass("Humanoid")
			local rig = hm and (hm.RigType == Enum.HumanoidRigType.R6 and "R6" or "R15") or "?"
			setStatus(("Alvo: %s | rig atual: %s"):format(player.Name, rig))
		end)

		y = y + 30
	end
	DropdownScroll.CanvasSize = UDim2.new(0, 0, 0, y)
end
